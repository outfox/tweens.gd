```csharp title="Godot's Tween"
var tween = sprite.CreateTween();
tween.TweenProperty(sprite, "position", new Vector2(400, 180), 0.6)
    .SetTrans(Tween.TransitionType.Cubic)
    .SetEase(Tween.EaseType.Out);

await ToSignal(tween, Tween.SignalName.Finished);
GD.Print("Arrived");
```

```csharp title="tweens.gd"
await sprite.TweenPosition((400, 180), 0.6, Out.Cubic).End;
GD.Print("Arrived");
```

<Compare>

```csharp title="Godot's Tween"
var tween1 = sprite1.CreateTween();
tween1.TweenProperty(sprite1, "position", new Vector2(400, 180), 0.6)
    .SetTrans(Tween.TransitionType.Cubic)
    .SetEase(Tween.EaseType.Out);

var tween2 = sprite2.CreateTween();
tween2.TweenProperty(sprite2, "position", new Vector2(400, 180), 0.8)
    .SetTrans(Tween.TransitionType.Cubic)
    .SetEase(Tween.EaseType.Out);

var tween3 = sprite3.CreateTween();
tween3.TweenProperty(sprite3, "position", new Vector2(400, 180), 1.0)
    .SetTrans(Tween.TransitionType.Cubic)
    .SetEase(Tween.EaseType.Out);

await ToSignal(tween3, Tween.SignalName.Finished);
GD.Print("Arrived");
```

```csharp title="tweens.gd"
var arrive = new Tweens.Position2D((400, 180), 0.6, Out.Cubic);

sprite1.Tween(arrive);
sprite2.Tween(arrive with { Duration = 0.8 });
await sprite3.Tween(arrive with { Duration = 1.0 }).End;
GD.Print("Arrived");
```

```gdscript title="Godot's Tween"
var tween := sprite.create_tween()
```

<Compare>

```gdscript title="Godot's Tween"
var tween1 := sprite1.create_tween()
tween1.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)
tween1.tween_property(sprite1, "position", Vector2(400, 180), 0.6)

var tween2 := sprite2.create_tween()
tween2.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)
tween2.tween_property(sprite2, "position", Vector2(400, 180), 0.8)

var tween3 := sprite3.create_tween()
tween3.set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_OUT)
tween3.tween_property(sprite3, "position", Vector2(400, 180), 1.0)

await tween3.finished
print("Arrived")
```

```gdscript title="tweens.gd"
var arrive := Tweens.position_2d([400, 180], 0.6, Out.CUBIC)

Tweens.play(sprite1, arrive)
Tweens.play(sprite2, arrive.with_duration(0.8))
await Tweens.play(sprite3, arrive.with_duration(1.0)).end
print("Arrived")
```
